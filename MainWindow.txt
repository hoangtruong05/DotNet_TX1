    public partial class MainWindow : Window
    {
        public MainWindow()
        {
            InitializeComponent();
        }
        QlbenhNhanContext db = new QlbenhNhanContext();
        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            HienThiBD();
            HienThiCB();
            //Scaffold-DbContext "Data Source=LAPTOP-PREP3IJM\SQLEXPRESS;Initial Catalog=QLBenhNhan;Integrated Security=True;Encrypt=True;Trust Server Certificate=True"
            //Microsoft.EntityFrameworkCore.SqlServer -OutputDir Models
        }

        private void HienThiBD()
        {
            dgHienThi.ItemsSource = db.BenhNhans.Where(x => x.SongayNv <=20)
                                                .OrderByDescending(x=>x.SongayNv)  
                                                .Select(x =>new
                                                 {
                                                     x.Mabn,
                                                     x.Hoten,
                                                     x.Diachi,
                                                     x.Makhoa,
                                                     x.SongayNv,
                                                     VienPhi = x.SongayNv * 60000
                                                 }).ToList();
        }

        private void HienThiCB()
        {
            cboKhoaKham.ItemsSource = db.KhoaKhams.Select(x =>x).ToList();
            cboKhoaKham.DisplayMemberPath = "Tenkhoa";
            cboKhoaKham.SelectedValuePath = "Makhoa";
            cboKhoaKham.SelectedIndex = 0;
        }
        private void dgBenhNhan_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            if (dgHienThi.SelectedItem != null)
            {
                dynamic b = dgHienThi.SelectedItem;

                txtMaBenhNhan.Text = b.Mabn;
                txtHoTen.Text = b.Hoten;
                txtDiaChi.Text = b.Diachi;
                txtSoNgayNamVien.Text = b.SongayNv.ToString();
                cboKhoaKham.SelectedValue = b.Makhoa;
            }
        }
        private bool isCheck()
        {
            if (txtMaBenhNhan.Text.Trim() == "" ||
               txtHoTen.Text.Trim() == "" ||
               txtDiaChi.Text.Trim() == "" ||
               txtSoNgayNamVien.Text.Trim() == "" ||
               cboKhoaKham.SelectedIndex == -1)
            {
                MessageBox.Show("Khong duoc de trong truong thong tin!!!", "Thong Bao", MessageBoxButton.OK, MessageBoxImage.Warning);
                return false;
            }
            if (!int.TryParse(txtSoNgayNamVien.Text.Trim(), out int soNgayNamVien) || soNgayNamVien <= 0)
            {
                MessageBox.Show("Vui long nhap lai so ngay nam vien!", "Thong bao", MessageBoxButton.OK, MessageBoxImage.Error);
                return false;
            }
            return true;
        }
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            var x = db.BenhNhans.SingleOrDefault(x => x.Mabn == txtMaBenhNhan.Text);
            if(x!=null)
            {
                MessageBox.Show("Ma thuoc da ton tai!!!");
                return;
            }

            else
            {
                try
                {
                    if (isCheck())
                    {
                        BenhNhan b = new BenhNhan();

                        b.Mabn = txtMaBenhNhan.Text;
                        b.Hoten = txtHoTen.Text;
                        b.Diachi = txtDiaChi.Text;
                        b.SongayNv = int.Parse(txtSoNgayNamVien.Text);
                        KhoaKham y = (KhoaKham)cboKhoaKham.SelectedItem;
                        b.Makhoa = y.Makhoa;
                        db.BenhNhans.Add(b);
                        db.SaveChanges();
                        HienThiBD();
                        //dgHienThi.Items.Refresh();
                        MessageBox.Show("Them benh nhan thanh cong!"); 
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Them benh nhan khong thanh cong!");
                }
            }
        }

        

        private void btnSua_Click(object sender, RoutedEventArgs e)
        {
            var x = db.BenhNhans.SingleOrDefault(x => x.Mabn == txtMaBenhNhan.Text);
            if (x == null)
            {
                MessageBox.Show("Khong tim thay benh nhan!!!");
                return;
            }

            else
            {
                try
                {
                    if (isCheck())
                    {
                                           
                        x.Hoten = txtHoTen.Text;
                        x.Diachi = txtDiaChi.Text;
                        x.SongayNv = int.Parse(txtSoNgayNamVien.Text);
                        KhoaKham y = (KhoaKham)cboKhoaKham.SelectedItem;
                        x.Makhoa = y.Makhoa;
                        db.SaveChanges();
                        HienThiBD();
                        MessageBox.Show("Sua benh nhan thanh cong!");
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Sua benh nhan khong thanh cong!");
                }
            }
        }

        private void btnXoa_Click(object sender, RoutedEventArgs e)
        {
            var bnXoa = db.BenhNhans.Find(txtMaBenhNhan.Text);
            if (bnXoa != null)
            {
                MessageBoxResult rs = MessageBox.Show("ban co chac chan muon xoa", "Xac nhan", MessageBoxButton.YesNo);
                if (rs == MessageBoxResult.Yes)
                {
                    db.Remove(bnXoa);
                    db.SaveChanges();
                    HienThiBD();
                }
            }
        }

        private void btnThongKe_Click(object sender, RoutedEventArgs e)
        {
            new ThongKe().Show();
        }
    }
}

ThongKe.xaml.cs

public partial class ThongKe : Window
{
    public ThongKe()
    {
        InitializeComponent();
    }
    QlbenhNhanContext db = new QlbenhNhanContext();
    private void Window_Loaded(object sender, RoutedEventArgs e)
    {
        dgHienThi.ItemsSource = (from bn in db.BenhNhans
                                    join kk in db.KhoaKhams
                                    on bn.Makhoa equals kk.Makhoa
                                    group bn by new {kk.Makhoa, kk.Tenkhoa} into g
                                    select new
                                    {
                                        MaKhoa = g.Key.Makhoa,
                                        TenKhoa = g.Key.Tenkhoa,
                                        TongVienPhi = g.Sum(bn => bn.SongayNv * 60000)
                                    }).ToList();
    }
