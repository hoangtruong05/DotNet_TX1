💻 Bước 2: Sửa đổi Logic xử lý C# MainWindow.xaml.cs
Bây giờ chúng ta sẽ lập trình thuật toán lấy ngày hiện tại trừ đi ngày nhập viện để tự tính toán số ngày lưu trú.  
TXT

1. Hàm hiển thị danh sách HienThiBD()
Chúng ta sử dụng cấu trúc động (Anonymous Type) sẵn có của bạn để chuyển đổi trực tiếp giá trị bool thành chuỗi "Nam"/"Nữ", đồng thời tính toán số ngày nằm viện bằng thuộc tính chênh lệch thời gian thời gian thực (DateTime.Now - x.NgayNhapVien).Days:  
TXT

C#
private void HienThiBD()
{
    dgHienThi.ItemsSource = db.BenhNhans
        // Bạn có thể giữ nguyên hoặc bỏ điều kiện Where tùy theo yêu cầu của đề mới
        .Where(x => (DateTime.Now - x.NgayNhapVien).Days <= 20) 
        .ToList() // Ép kiểu về List trước để xử lý chuỗi và tính toán mượt mà trên bộ nhớ RAM
        .Select(x => new
        {
            x.Mabn,
            x.Hoten,
            x.Diachi,
            x.Makhoa,
            NgayNhapVien = x.NgayNhapVien,
            // Thuật toán: Lấy ngày hiện tại trừ ngày nhập viện, dùng .Days để lấy ra số nguyên số ngày
            SongayNv = (DateTime.Now - x.NgayNhapVien).Days <= 0 ? 1 : (DateTime.Now - x.NgayNhapVien).Days, 
            // Chuyển đổi dữ liệu true/false thành chữ Nam/Nữ hiển thị trực quan lên bảng
            GioiTinhHienThi = x.GioiTinh == true ? "Nam" : "Nữ",
            // Tính toán viện phí động dựa trên số ngày vừa tính ra ở trên
            VienPhi = ((DateTime.Now - x.NgayNhapVien).Days <= 0 ? 1 : (DateTime.Now - x.NgayNhapVien).Days) * 60000
        })
        [cite_start].OrderByDescending(x => x.SongayNv) // Đảm bảo sắp xếp giảm dần theo số ngày nằm viện thực tế [cite: 16]
        .ToList();
}
2. Sự kiện Click dòng bảng dgBenhNhan_SelectionChanged
Khi người dùng chọn dòng , chúng ta hiển thị ngược lại trạng thái tích chọn cho RadioButton và ô chọn lịch DatePicker:  


C#
private void dgBenhNhan_SelectionChanged(object sender, SelectionChangedEventArgs e)
{
    if (dgHienThi.SelectedItem != null)
    {
        dynamic b = dgHienThi.SelectedItem;
        txtMaBenhNhan.Text = b.Mabn;
        txtHoTen.Text = b.Hoten;
        txtDiaChi.Text = b.Diachi;
        
        // Hiển thị ngày nhập viện lên ô chọn lịch
        dtpNgayNhapVien.SelectedDate = b.NgayNhapVien;
        
        // Hiển thị giới tính lên RadioButton
        if (b.GioiTinhHienThi == "Nam")
            rdoNam.IsChecked = true;
        else
            rdoNu.IsChecked = true;

        cboKhoaKham.SelectedValue = b.Makhoa;
    }
}
3. Hàm kiểm tra dữ liệu đầu vào isCheck()
Vì số ngày nằm viện giờ đây đã được tự động tính toán bằng lịch, bạn không cần kiểm tra số ngày ở ô TextBox nữa. Thay vào đó, bạn chỉ cần kiểm tra xem người dùng đã chọn ngày nhập viện trên lịch hay chưa:  


C#
private bool isCheck()
{
    if (txtMaBenhNhan.Text.Trim() == "" ||
        txtHoTen.Text.Trim() == "" ||
        txtDiaChi.Text.Trim() == "" ||
        dtpNgayNhapVien.SelectedDate == null || // Kiểm tra lịch không được bỏ trống
        cboKhoaKham.SelectedIndex == -1)
    {
        MessageBox.Show("Không được để trống trường thông tin!!!", "Thông Báo", MessageBoxButton.OK, MessageBoxImage.Warning);
        return false;
    }
    
    if (dtpNgayNhapVien.SelectedDate > DateTime.Now)
    {
        MessageBox.Show("Ngày nhập viện không thể lớn hơn ngày hiện tại!", "Thông báo", MessageBoxButton.OK, MessageBoxImage.Error);
        return false;
    }
    return true;
}
